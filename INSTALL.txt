================================================================================
  SUBZ STUDIO PRO — FULL VPS INSTALL GUIDE
  Production-Ready Setup for Ubuntu 22.04 / 24.04 LTS
  studio.subz.lk
================================================================================

WHAT THIS INSTALLS
------------------
  - Flask/Socket.IO subtitle-burning pipeline (port 5000)
  - Gunicorn + Gevent (production WSGI server)
  - Nginx reverse proxy with SSL (Let's Encrypt)
  - Telegram Local Bot API Server (port 8081, up to 2GB uploads)
  - FFmpeg with libass (Sinhala subtitle burn-in)
  - MySQL 8.0 (encode/upload history)
  - Systemd services (auto-start on reboot)
  - Fail2ban (brute force protection)
  - UFW firewall

REQUIREMENTS
------------
  - Fresh Ubuntu 22.04 or 24.04 VPS (min 2 CPU, 4GB RAM, 50GB disk)
  - Root or sudo access
  - Domain pointing to your VPS IP (e.g. studio.subz.lk → A record)
  - Telegram Bot Token (from @BotFather)
  - Telegram API ID + Hash (from https://my.telegram.org/apps)

================================================================================
  STEP 7 — SYSTEMD SERVICES
================================================================================

---- 7A. Subz Studio App Service ----

sudo tee /etc/systemd/system/subz-studio.service << 'SERVICE'
[Unit]
Description=Subz Studio Pro — Subtitle Burning Pipeline
After=network.target mysql.service
Wants=mysql.service

[Service]
Type=simple
User=subz
Group=subz
WorkingDirectory=/home/subz/subz-studio
EnvironmentFile=/home/subz/subz-studio/.env
ExecStart=/home/subz/subz-studio/venv/bin/gunicorn \
    --worker-class gevent \
    --workers 1 \
    --timeout 1800 \
    --bind 127.0.0.1:5000 \
    --log-level info \
    --access-logfile /home/subz/subz-studio/logs/access.log \
    --error-logfile  /home/subz/subz-studio/logs/error.log \
    app:app
Restart=always
RestartSec=5
StandardOutput=journal
StandardError=journal
KillMode=mixed
TimeoutStopSec=30

[Install]
WantedBy=multi-user.target
SERVICE

---- 7B. Telegram Bot API Server Service ----

sudo tee /etc/systemd/system/tg-bot-api.service << 'SERVICE'
[Unit]
Description=Telegram Local Bot API Server
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=subz
Group=subz
ExecStart=/usr/local/bin/telegram-bot-api \
    --api-id=36668698 \
    --api-hash=5e1172b296563abf8ba9939c557c9f66 \
    --local \
    --http-port=8081 \
    --log=/home/subz/subz-studio/logs/tgapi.log \
    --verbosity=1
Restart=always
RestartSec=10
StandardOutput=journal

[Install]

================================================================================
  STEP 8 — NGINX CONFIGURATION
================================================================================

sudo tee /etc/nginx/sites-available/studio.subz.lk << 'NGINX'
upstream subz_app {
    server 127.0.0.1:5000;
    keepalive 32;
}

server {
    listen 80;
    server_name studio.subz.lk;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    server_name studio.subz.lk;

    # SSL (filled by Certbot below)
    ssl_certificate     /etc/letsencrypt/live/studio.subz.lk/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/studio.subz.lk/privkey.pem;
    include             /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam         /etc/letsencrypt/ssl-dhparams.pem;

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN"          always;
    add_header X-Content-Type-Options "nosniff"      always;
    add_header X-XSS-Protection "1; mode=block"      always;
    add_header Referrer-Policy "strict-origin"        always;

    # Large file uploads (2GB for Telegram)
    client_max_body_size 0;
    proxy_read_timeout   3600s;
    proxy_send_timeout   3600s;
    proxy_connect_timeout 60s;

    # Gzip
    gzip on;
    gzip_types text/plain application/json text/css application/javascript;

    # Main app
    location / {
        proxy_pass         http://subz_app;
        proxy_http_version 1.1;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;

        # Socket.IO WebSocket support
        proxy_set_header   Upgrade    $http_upgrade;
        proxy_set_header   Connection "upgrade";
        proxy_buffering    off;
        proxy_cache        off;
    }

    # Direct download — serve from disk, bypass app
    location /download/ {
        alias /home/subz/subz-studio/downloads/;
        add_header Content-Disposition "attachment";
        sendfile on;
        tcp_nopush on;
        aio on;
    }

    # Logs
    access_log /var/log/nginx/studio.subz.lk.access.log;
    error_log  /var/log/nginx/studio.subz.lk.error.log;
}
NGINX

# Enable site
sudo ln -sf /etc/nginx/sites-available/studio.subz.lk /etc/nginx/sites-enabled/
sudo nginx -t && sudo systemctl reload nginx

================================================================================
  STEP 10 — FIREWALL (UFW)
================================================================================

sudo ufw default deny incoming
sudo ufw default allow outgoing
sudo ufw allow ssh
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
# Do NOT open 5000 or 8081 publicly — Nginx proxies 5000, 8081 is internal only
sudo ufw enable
sudo ufw status

================================================================================

================================================================================

sudo tee /etc/fail2ban/jail.local << 'F2B'
[DEFAULT]
bantime  = 3600
findtime = 600
maxretry = 5

[sshd]
enabled = true
port    = ssh
logpath = /var/log/auth.log

[nginx-http-auth]
enabled = true

[nginx-limit-req]
enabled = true
port    = http,https
logpath = /var/log/nginx/*.error.log
F2B

sudo systemctl restart fail2ban
sudo fail2ban-client status

================================================================================
  TROUBLESHOOTING
================================================================================

PROBLEM: "libass not found" or subtitles not burning
FIX:
  ffmpeg -filters 2>&1 | grep ass
  # If empty: apt-get install --reinstall ffmpeg
  # Or build FFmpeg from source with --enable-libass

PROBLEM: Sinhala font not rendering (boxes/squares)
FIX:
  fc-list | grep -i nirmala
  # If empty: font didn't copy. Check /home/subz/subz-studio/fonts/
  fc-cache -fv /home/subz/subz-studio/fonts/
  # Restart app: sudo systemctl restart subz-studio

PROBLEM: Telegram upload fails "Connection refused"
FIX:
  sudo systemctl status tg-bot-api
  sudo systemctl restart tg-bot-api
  curl http://localhost:8081/botTOKEN/getMe
  # If "not installed": run install-telegram-bot-api.sh

PROBLEM: MySQL connection failed
FIX:
  mysql -u subz_user -p subz_studio
  # Check .env DB_PASS matches MySQL user password
  # Check: sudo systemctl status mysql

PROBLEM: Socket.IO not connecting (progress bar not updating)
FIX:
  # Nginx must pass Upgrade header — check nginx config has:
  # proxy_set_header Upgrade $http_upgrade;
  # proxy_set_header Connection "upgrade";
  sudo nginx -t && sudo systemctl reload nginx

PROBLEM: Large videos fail (timeout)
FIX:
  # Increase gunicorn timeout in systemd service:
  # --timeout 3600
  # Nginx: proxy_read_timeout 3600s;
  sudo systemctl daemon-reload && sudo systemctl restart subz-studio
